213	  const submittedHash = crypto.createHash('sha256').update(String(submittedPin)).digest();

219	function firewallValidate(socket, cost = 1) {
220	
221	  const now = Date.now();
222	
223	  let client = clientFirewallState.get(socket.id);
224	
225	
226	
227	  if (!client) {
228	
229	    client = { tokens: 10, lastRefill: now, authorized: false };
230	
231	    clientFirewallState.set(socket.id, client);
232	
233	  }
234	
235	
236	
237	  const elapsed = (now - client.lastRefill) / 1000;
238	
239	  client.tokens = Math.min(10, client.tokens + elapsed * 3);
240	
241	  client.lastRefill = now;
242	
243	
244	
245	  if (client.tokens < cost) {
246	
247	    totalThreatsBlocked++;
248	
249	    socket.emit('firewall:alert', {
250	
251	      type: 'RATE_LIMIT_EXCEEDED',
252	
253	      msg: 'Command rate limit exceeded. Action dropped.'
254	
255	    });
256	
257	    return false;
258	
259	  }
260	
261	
262	
263	  client.tokens -= cost;
264	
265	  return true;
266	
267	}

333	async function fetchSatelliteWeather(key = 'setif') {
334	
335	  try {
336	
337	    if (!Object.prototype.hasOwnProperty.call(LOCATION_COORDINATES, key)) {
338	
339	      key = 'setif';
340	
341	    }
342	
343	    const loc = LOCATION_COORDINATES[key];
344	
345	    activeLocationKey = key;
346	
347	
348	
349	    const url = new URL('https://api.open-meteo.com/v1/forecast');
350	
351	    url.searchParams.set('latitude', loc.lat.toString());
352	
353	    url.searchParams.set('longitude', loc.lon.toString());
354	
355	    url.searchParams.set('current', 'temperature_2m,relative_humidity_2m,precipitation,wind_speed_10m,weather_code');
356	
357	    url.searchParams.set('hourly', 'precipitation_probability,precipitation,temperature_2m,wind_speed_10m');
358	
359	    url.searchParams.set('daily', 'et0_fao_evapotranspiration');
360	
361	    url.searchParams.set('forecast_days', '2');
362	
363	    url.searchParams.set('timezone', 'auto');
364	
365	    
366	
367	    const response = await fetch(url.toString(), { signal: AbortSignal.timeout(6000) });
368	
369	    if (!response.ok) throw new Error(`Open-Meteo HTTP ${response.status}`);
370	
371	    
372	
373	    const data = await response.json();
374	
375	    const cur = data.current || {};
376	
377	    const daily = data.daily || {};
378	
379	    const hourly = data.hourly || {};
380	
381	
382	
383	    const liveData = {
384	
385	      location: loc.name,
386	
387	      temp: typeof cur.temperature_2m === 'number' ? cur.temperature_2m : 24.0,
388	
389	      humidity: typeof cur.relative_humidity_2m === 'number' ? cur.relative_humidity_2m : 50,
390	
391	      rain: typeof cur.precipitation === 'number' ? cur.precipitation : 0.0,
392	
393	      windSpeed: typeof cur.wind_speed_10m === 'number' ? cur.wind_speed_10m : 8.0,
394	
395	      weatherCode: typeof cur.weather_code === 'number' ? cur.weather_code : 0,
396	
397	      et0: (Array.isArray(daily.et0_fao_evapotranspiration) && daily.et0_fao_evapotranspiration.length > 0)
398	
399	        ? Number(daily.et0_fao_evapotranspiration[0])
400	
401	        : 4.2
402	
403	    };
404	
405	
406	
407	    simulator.setLiveWeather(liveData);
408	
409	
410	
411	    if (hourly.time && Array.isArray(hourly.time)) {
412	
413	      const currentHourStr = cur.time ? cur.time.substring(0, 13) : '';
414	
415	      let startIndex = hourly.time.findIndex(t => t.startsWith(currentHourStr));
416	
417	      if (startIndex < 0) startIndex = 0;
418	
419	
420	
421	      const horizon24 = [];
422	
423	      let maxRainProb12h = 0;
424	
425	      let totalRain24h = 0;
426	
427	
428	
429	      for (let i = 0; i < 24; i++) {
430	
431	        const idx = startIndex + i;
432	
433	        if (idx < hourly.time.length) {
434	
435	          const timeLabel = hourly.time[idx].split('T')[1] || `${i}:00`;
436	
437	          const prob = hourly.precipitation_probability ? Number(hourly.precipitation_probability[idx]) || 0 : 0;
438	
439	          const rainMm = hourly.precipitation ? Number(hourly.precipitation[idx]) || 0 : 0;
440	
441	          const temp = hourly.temperature_2m ? Number(hourly.temperature_2m[idx]) || 20 : 20;
442	
443	
444	
445	          if (i <= 12 && prob > maxRainProb12h) maxRainProb12h = prob;
446	
447	          totalRain24h += rainMm;
448	
449	
450	
451	          horizon24.push({ hour: timeLabel, prob, rainMm, temp });
452	
453	        }
454	
455	      }
456	
457	
458	
459	      const willRainSoon = maxRainProb12h >= 60 || totalRain24h >= 4.0;
460	
461	      const expectedSaving = willRainSoon ? Math.round(1800 + totalRain24h * 450) : 0;
462	
463	
464	
465	      latestAIPrediction = {
466	
467	        horizon: horizon24,
468	
469	        rainHoldActive: willRainSoon,
470	
471	        maxRainProb12h,
472	
473	        totalRain24h: Number(totalRain24h.toFixed(1)),
474	
475	        confidence: willRainSoon ? Math.min(98, 80 + Math.round(maxRainProb12h * 0.2)) : 92,
476	
477	        recommendation: willRainSoon ? 'AUTONOMOUS_RAIN_HOLD' : 'NOMINAL_IRRIGATION',
478	
479	        waterSavedEstimateL: expectedSaving,
480	
481	        rationale: willRainSoon
482	
483	          ? `Precipitation front detected (Peak: ${maxRainProb12h}%, Cumul: ${totalRain24h.toFixed(1)}mm). Pump dispatch postponed to leverage natural precipitation.`
484	
485	          : `Clear atmospheric outlook across next 24h. Soil moisture depletion will follow standard PID setpoint.`
486	
487	      };
488	
489	    }
490	
491	  } catch (err) {
492

132	function generateVibrationWaveform(targetRms, isRunning, pwm, bearingWear, cavitation) {
133	  if (!isRunning || targetRms < 0.05) {
134	    const idleSamples = new Array(WAVEFORM_SAMPLES);
135	    for (let i = 0; i < WAVEFORM_SAMPLES; i++) {
136	      idleSamples[i] = Number(((Math.random() - 0.5) * 0.06).toFixed(3));
137	    }
138	    return { samples: idleSamples, dominantFreq: 0.0 };
139	  }
140	
141	  const f0 = 45.0 + (pwm / 100.0) * 10.0;
142	  const dt = 1.0 / SAMPLING_RATE_HZ;
143	  const fBearing = f0 * 3.56;
144	  const bearingSeverity = bearingWear / 100.0;
145	  const cavitationFactor = Math.max(0, (cavitation - 10.0) / 90.0);
146	
147	  let sumSquares = 0.0;
148	  const rawSignal = new Float64Array(WAVEFORM_SAMPLES);
149	
150	  for (let n = 0; n < WAVEFORM_SAMPLES; n++) {
151	    const t = n * dt;
152	
153	    let s = Math.sin(2 * Math.PI * f0 * t);
154	    s += 0.35 * Math.sin(2 * Math.PI * (2 * f0) * t + 0.4);
155	
156	    if (bearingSeverity > 0.04) {
157	      const impact = Math.sin(2 * Math.PI * fBearing * t);
158	      s += (bearingSeverity * 2.2) * impact * (1.0 + 0.5 * Math.sin(2 * Math.PI * f0 * t));
159	    }
160	
161	    if (cavitationFactor > 0.0) {
162	      s += (cavitationFactor * 2.5) * (Math.random() - 0.5);
163	    }
164	
165	    s += 0.12 * (Math.random() - 0.5);
166	
167	    rawSignal[n] = s;
168	    sumSquares += s * s;
169	  }
170	
171	  const currentRms = Math.sqrt(sumSquares / WAVEFORM_SAMPLES) || 1.0;
172	  const scale = targetRms / currentRms;
173	
174	  const finalSamples = new Array(WAVEFORM_SAMPLES);
175	  for (let n = 0; n < WAVEFORM_SAMPLES; n++) {
176	    finalSamples[n] = Number((rawSignal[n] * scale).toFixed(3));
177	  }
178	
179	  return {
180	    samples: finalSamples,
181	    dominantFreq: Number(f0.toFixed(1))
182	  };
183	}

326	function pidLoop() {
327	  // صمام تعبئة الخزان الآلي (Hysteresis): لا يفتح إلا عند الهبوط تحت 12% ويغلق عند 80%
328	  if (state.tankVolumePct <= 12.0) {

331	    state.tankInletValve = false;
332	  }
333	
334	  const activeInflow = state.tankInletValve ? 0.35 : TANK_RECHARGE_RATE;
335	  const waterConsumedL = (state.flowRate / 60.0) * DT;
336	
337	  // تناقص مستوى الخزان الحقيقي ثانية بثانية
338	  state.tankVolumeL = Math.max(0, state.tankVolumeL - waterConsumedL + (activeInflow * DT));
339	  state.tankVolumeL = Math.min(state.tankCapacityL, state.tankVolumeL);
340	  updateReservoirCavitation();
341	
342	  const activePlot = state.zones.find(z => z.id === state.activeZoneId);
343	  if (activePlot && waterConsumedL > 0) {
344	    activePlot.waterUsedL = Number((activePlot.waterUsedL + waterConsumedL).toFixed(2));
345	  }
346	
347	  updateMotorThermalModel();
348	  checkAntiFlooding();
349	
350	  let rawPwmCommand = 0.0;
351	  if (state.isManual) {
352	    rawPwmCommand = state.manualPwm;
353	  } else {
354	    state.error = state.setpoint - state.vwc;
355	    state.integralAcc += state.error * DT;
356	    state.integralAcc = Math.max(-25.0, Math.min(25.0, state.integralAcc));
357	
358	    const derivative = (state.error - state.lastError) / DT;
359	    state.lastError = state.error;
360	
361	    state.pTerm = state.kp * state.error;
362	    state.iTerm = state.ki * state.integralAcc;
363	    state.dTerm = state.kd * derivative;
364	
365	    let computed = state.pTerm + state.iTerm + state.dTerm;
366	    rawPwmCommand = Math.max(0.0, Math.min(100.0, computed));
367	  }
368	
369	  // Interlock overrides
370	  state.activeFaults = [];
371	  if (state.interlocks.dryRun) {
372	    rawPwmCommand = 0.0;
373	    state.activeFaults.push('LOCK: RESERVOIR CAVITATION PREVENTED (LEVEL < 5%)');
374	  }
375	  if (state.interlocks.pipeBurst) {
376	    rawPwmCommand = 0.0;
377	    state.activeFaults.push('LOCK: MAJOR HYDRAULIC PIPE RUPTURE DETECTED');
378	  }
379	  if (state.interlocks.floodRisk) {
380	    rawPwmCommand = 0.0;
381	    state.activeFaults.push('LOCK: ANTI-FLOODING / ROOT ASPHYXIATION OVERRIDE');
382	  }
383	  if (state.interlocks.thermalTrip) {
384	    rawPwmCommand = Math.min(rawPwmCommand, 30.0);
385	    state.activeFaults.push('THROTTLE: MOTOR COIL THERMAL DERATING ACTIVE (TEMP > 85°C)');
386	  }
387	
388	  if (state.interlocks.dryRun || state.interlocks.pipeBurst || state.interlocks.floodRisk) {

13	// 1. خوارزمية حساب متطلبات الري ومنع الهدر المناخي
14	function calculateIrrigationDuty(currentVwc, setpoint, rainProb) {
15	  var error = setpoint - currentVwc;
16	
17	  // في حال وجود احتمالية أمطار تفوق 60% يتم تأجيل الضخ تلقائياً
18	  if (rainProb >= 60) {
19	    return { duty: 0, mode: "AUTONOMOUS_RAIN_HOLD" };
20	  }
21	
22	  // حساب نسبة الضخ حسب فارق الرطوبة
23	  var duty = Math.max(0, Math.min(100, error * 2.2));
24	  return { duty: Math.round(duty), mode: "CLOSED_LOOP_ACTIVE" };
25	}

27	// 2. صمامات الأمان الصناعي وحماية المضخة (ISO 10816 Severity Baseline)
28	function evaluateSafety(telemetry) {
29	  let actions = {
30	    tripPump: false,          // إيقاف المضخة التلقائي في الحالات الخطرة
31	    systemStatus: 'NORMAL',   // NORMAL | WARNING | CRITICAL
32	    alarms: []
33	  };
34	
35	  // أ. حماية من التشغيل الجاف (Dry-Run Protection مع فلترة التأخير الزمني)
36	  // لا نوقف المضخة إلا إذا استمرت بالعمل بحمل حقيقي (>15%) بدون تدفق لمدة 4 ثوانٍ متتالية
37	  if (telemetry.pumpState && telemetry.pumpDuty > 15.0 && telemetry.flowRate <= 0.8) {
38	    dryRunDebounceCounter++;
39	    if (dryRunDebounceCounter >= 4) {
40	      actions.tripPump = true;
41	      actions.systemStatus = 'CRITICAL';
42	      actions.alarms.push('CRITICAL: Dry-run verified. Zero flow while pump running under load.');
43	    } else {
44	      actions.systemStatus = 'WARNING';
45	      actions.alarms.push(`WARNING: Establishing hydraulic flow (${dryRunDebounceCounter}/4s)...`);
46	    }
47	  } else {
48	    dryRunDebounceCounter = Math.max(0, dryRunDebounceCounter - 1);
49	  }
50	
51	  // ب. حماية المحرك من ارتفاع درجة الحرارة (Thermal Overheat)
52	  if (telemetry.motorTemp >= 85.0) {
53	    actions.tripPump = true;
54	    actions.systemStatus = 'CRITICAL';
55	    actions.alarms.push(`CRITICAL: Motor overheat (${telemetry.motorTemp}°C >= 85°C).`);
56	  } else if (telemetry.motorTemp >= 75.0) {
57	    actions.systemStatus = actions.systemStatus === 'CRITICAL' ? 'CRITICAL' : 'WARNING';
58	    actions.alarms.push(`WARNING: High motor temperature (${telemetry.motorTemp}°C).`);
59	  }
60	
61	  // ج. معيار الاهتزاز الميكانيكي وتآكل المحامل (ISO 10816 Class II)
62	  // Class II thresholds: Warning >= 2.8 mm/s, Critical Trip >= 7.1 mm/s
63	  if (telemetry.vibrationRms >= 7.1) {
64	    actions.tripPump = true;
65	    actions.systemStatus = 'CRITICAL';
66	    actions.alarms.push(`CRITICAL: Unacceptable vibration (${telemetry.vibrationRms} mm/s). Emergency stop.`);
67	  } else if (telemetry.vibrationRms >= 2.8) {
68	    actions.systemStatus = actions.systemStatus === 'CRITICAL' ? 'CRITICAL' : 'WARNING';
69	    actions.alarms.push(`WARNING: Unsatisfactory vibration (${telemetry.vibrationRms} mm/s). Maintenance needed.`);
70	  }
71	
72	  return actions;
73	}

152	    if (audioMuted || !audioCtx) return;
153	
154	    try {
155	      var osc = audioCtx.createOscillator();
156	      var gain = audioCtx.createGain();
157	
158	      osc.type = type || "sine";
159	      osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
160	
161	      gain.gain.setValueAtTime(
162	        vol || 0.1,
163	        audioCtx.currentTime
164	      );
165	
166	      gain.gain.exponentialRampToValueAtTime(
167	        0.001,
168	        audioCtx.currentTime + duration
169	      );
170	
171	      osc.connect(gain);
172	      gain.connect(audioCtx.destination);
173	
174	      osc.start();
175	      osc.stop(audioCtx.currentTime + duration);
176	    } catch (e) {}
177	  }

2490	    usbReader =
2491	      reader;

78	  // حساب الـ Spectrum مع معايرة السعة الفيزيائية الصحيحة (Single-Sided Scaled DFT)
79	  function computeFFT(samples) {
80	    var N = samples.length;
81	    var magnitudes = [];
82	    var halfN = Math.floor(N / 2);
83	
84	    for (var k = 0; k < halfN; k++) {
85	      var real = 0;
86	      var imag = 0;
87	      for (var n = 0; n < N; n++) {
88	        var angle = (2 * Math.PI * k * n) / N;
89	        real += samples[n] * Math.cos(angle);
90	        imag -= samples[n] * Math.sin(angle);
91	      }
92	      var mag = Math.sqrt(real * real + imag * imag);
93	      // التردد الصفري DC يقسم على N، وبقية الترددات تضرب في 2/N لمعايرة السعة الفيزيائية الحقيقية
94	      var scaled = (k === 0) ? (mag / N) : ((2 * mag) / N);
95	      magnitudes.push(Number(scaled.toFixed(4)));
96	    }
97	    return magnitudes;
98	  }
