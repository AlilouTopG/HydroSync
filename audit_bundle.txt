215	function firewallValidate(socket, cost = 1) {
216	
217	  const now = Date.now();
218	
219	  let client = clientFirewallState.get(socket.id);
220	
221	
222	
223	  if (!client) {
224	
225	    client = { tokens: 10, lastRefill: now, authorized: false };
226	
227	    clientFirewallState.set(socket.id, client);
228	
229	  }
230	
231	
232	
233	  const elapsed = (now - client.lastRefill) / 1000;
234	
235	  client.tokens = Math.min(10, client.tokens + elapsed * 3);
236	
237	  client.lastRefill = now;
238	
239	
240	
241	  if (client.tokens < cost) {
242	
243	    totalThreatsBlocked++;
244	
245	    socket.emit('firewall:alert', {
246	
247	      type: 'RATE_LIMIT_EXCEEDED',
248	
249	      msg: 'Command rate limit exceeded. Action dropped.'
250	
251	    });
252	
253	    return false;
254	
255	  }
256	
257	
258	
259	  client.tokens -= cost;
260	
261	  return true;
262	
263	}

329	async function fetchSatelliteWeather(key = 'setif') {
330	
331	  try {

333	    if (!Object.prototype.hasOwnProperty.call(LOCATION_COORDINATES, key)) {

342	
343	
344	
345	    const url = new URL('https://api.open-meteo.com/v1/forecast');
346	
347	    url.searchParams.set('latitude', loc.lat.toString());
348	
349	    url.searchParams.set('longitude', loc.lon.toString());
350	
351	    url.searchParams.set('current', 'temperature_2m,relative_humidity_2m,precipitation,wind_speed_10m,weather_code');
352	
353	    url.searchParams.set('hourly', 'precipitation_probability,precipitation,temperature_2m,wind_speed_10m');
354	
355	    url.searchParams.set('daily', 'et0_fao_evapotranspiration');
356	
357	    url.searchParams.set('forecast_days', '2');
358	
359	    url.searchParams.set('timezone', 'auto');
360	
361	    
362	
363	    const response = await fetch(url.toString(), { signal: AbortSignal.timeout(6000) });
364	
365	    if (!response.ok) throw new Error(`Open-Meteo HTTP ${response.status}`);
366	
367	    
368	
369	    const data = await response.json();
370	
371	    const cur = data.current || {};
372	
373	    const daily = data.daily || {};
374	
375	    const hourly = data.hourly || {};
376	
377	
378	
379	    const liveData = {
380	
381	      location: loc.name,
382	
383	      temp: typeof cur.temperature_2m === 'number' ? cur.temperature_2m : 24.0,
384	
385	      humidity: typeof cur.relative_humidity_2m === 'number' ? cur.relative_humidity_2m : 50,
386	
387	      rain: typeof cur.precipitation === 'number' ? cur.precipitation : 0.0,
388	
389	      windSpeed: typeof cur.wind_speed_10m === 'number' ? cur.wind_speed_10m : 8.0,
390	
391	      weatherCode: typeof cur.weather_code === 'number' ? cur.weather_code : 0,
392	
393	      et0: (Array.isArray(daily.et0_fao_evapotranspiration) && daily.et0_fao_evapotranspiration.length > 0)
394	
395	        ? Number(daily.et0_fao_evapotranspiration[0])
396	
397	        : 4.2
398	
399	    };
400	
401	
402	
403	    simulator.setLiveWeather(liveData);
404	
405	
406	
407	    if (hourly.time && Array.isArray(hourly.time)) {
408	
409	      const currentHourStr = cur.time ? cur.time.substring(0, 13) : '';
410	
411	      let startIndex = hourly.time.findIndex(t => t.startsWith(currentHourStr));
412	
413	      if (startIndex < 0) startIndex = 0;
414	
415	
416	
417	      const horizon24 = [];
418	
419	      let maxRainProb12h = 0;
420	
421	      let totalRain24h = 0;
422	
423	
424	
425	      for (let i = 0; i < 24; i++) {
426	
427	        const idx = startIndex + i;
428	
429	        if (idx < hourly.time.length) {
430	
431	          const timeLabel = hourly.time[idx].split('T')[1] || `${i}:00`;
432	
433	          const prob = hourly.precipitation_probability ? Number(hourly.precipitation_probability[idx]) || 0 : 0;
434	
435	          const rainMm = hourly.precipitation ? Number(hourly.precipitation[idx]) || 0 : 0;
436	
437	          const temp = hourly.temperature_2m ? Number(hourly.temperature_2m[idx]) || 20 : 20;
438	
439	
440	
441	          if (i <= 12 && prob > maxRainProb12h) maxRainProb12h = prob;
442	
443	          totalRain24h += rainMm;
444	
445	
446	
447	          horizon24.push({ hour: timeLabel, prob, rainMm, temp });
448	
449	        }
450	
451	      }
452	
453	
454	
455	      const willRainSoon = maxRainProb12h >= 60 || totalRain24h >= 4.0;
456	
457	      const expectedSaving = willRainSoon ? Math.round(1800 + totalRain24h * 450) : 0;
458	
459	
460	
461	      latestAIPrediction = {
462	
463	        horizon: horizon24,
464	
465	        rainHoldActive: willRainSoon,
466	
467	        maxRainProb12h,
468	
469	        totalRain24h: Number(totalRain24h.toFixed(1)),
470	
471	        confidence: willRainSoon ? Math.min(98, 80 + Math.round(maxRainProb12h * 0.2)) : 92,
472	
473	        recommendation: willRainSoon ? 'AUTONOMOUS_RAIN_HOLD' : 'NOMINAL_IRRIGATION',
474	
475	        waterSavedEstimateL: expectedSaving,
476	
477	        rationale: willRainSoon

479	          ? `Precipitation front detected (Peak: ${maxRainProb12h}%, Cumul: ${totalRain24h.toFixed(1)}mm). Pump dispatch postponed to leverage natural precipitation.`

135	function generateVibrationWaveform(targetRms, isRunning, pwm, bearingWear, cavitation) {
136	  if (!isRunning || targetRms < 0.05) {
137	    const idleSamples = new Array(WAVEFORM_SAMPLES);
138	    for (let i = 0; i < WAVEFORM_SAMPLES; i++) {
139	      idleSamples[i] = Number(((Math.random() - 0.5) * 0.06).toFixed(3));
140	    }
141	    return { samples: idleSamples, dominantFreq: 0.0 };
142	  }
143	
144	  const f0 = 45.0 + (pwm / 100.0) * 10.0;
145	  const dt = 1.0 / SAMPLING_RATE_HZ;
146	  const fBearing = f0 * 3.56;
147	  const bearingSeverity = bearingWear / 100.0;
148	  const cavitationFactor = Math.max(0, (cavitation - 10.0) / 90.0);
149	
150	  let sumSquares = 0.0;
151	  const rawSignal = new Float64Array(WAVEFORM_SAMPLES);
152	
153	  for (let n = 0; n < WAVEFORM_SAMPLES; n++) {
154	    const t = n * dt;
155	
156	    let s = Math.sin(2 * Math.PI * f0 * t);
157	    s += 0.35 * Math.sin(2 * Math.PI * (2 * f0) * t + 0.4);
158	
159	    if (bearingSeverity > 0.04) {
160	      const impact = Math.sin(2 * Math.PI * fBearing * t);
161	      s += (bearingSeverity * 2.2) * impact * (1.0 + 0.5 * Math.sin(2 * Math.PI * f0 * t));
162	    }
163	
164	    if (cavitationFactor > 0.0) {
165	      s += (cavitationFactor * 2.5) * (Math.random() - 0.5);
166	    }
167	
168	    s += 0.12 * (Math.random() - 0.5);
169	
170	    rawSignal[n] = s;
171	    sumSquares += s * s;
172	  }
173	
174	  const currentRms = Math.sqrt(sumSquares / WAVEFORM_SAMPLES) || 1.0;
175	  const scale = targetRms / currentRms;
176	
177	  const finalSamples = new Array(WAVEFORM_SAMPLES);
178	  for (let n = 0; n < WAVEFORM_SAMPLES; n++) {
179	    finalSamples[n] = Number((rawSignal[n] * scale).toFixed(3));
180	  }
181	
182	  return {
183	    samples: finalSamples,
184	    dominantFreq: Number(f0.toFixed(1))
185	  };
186	}

331	function pidLoop() {
332	  // صمام تعبئة الخزان الآلي (Hysteresis): لا يفتح إلا عند الهبوط تحت 12% ويغلق عند 80%
333	  if (state.tankVolumePct <= 12.0) {
334	    state.tankInletValve = true;
335	  } else if (state.tankVolumePct >= 80.0) {
336	    state.tankInletValve = false;
337	  }
338	
339	  const activeInflow = state.tankInletValve ? 0.35 : TANK_RECHARGE_RATE;
340	  const waterConsumedL = (state.flowRate / 60.0) * DT;
341	
342	  // تناقص مستوى الخزان الحقيقي ثانية بثانية
343	  state.tankVolumeL = Math.max(0, state.tankVolumeL - waterConsumedL + (activeInflow * DT));
344	  state.tankVolumeL = Math.min(state.tankCapacityL, state.tankVolumeL);
345	  updateReservoirCavitation();
346	
347	  const activePlot = state.zones.find(z => z.id === state.activeZoneId);
348	  if (activePlot && waterConsumedL > 0) {
349	    activePlot.waterUsedL = Number((activePlot.waterUsedL + waterConsumedL).toFixed(2));
350	  }
351	
352	  updateMotorThermalModel();
353	  checkAntiFlooding();
354	
355	  let rawPwmCommand = 0.0;
356	  if (state.isManual) {
357	    rawPwmCommand = state.manualPwm;
358	  } else {
359	    state.error = state.setpoint - state.vwc;
360	    const holding = state.rainHold && state.vwc >= state.setpoint - RAIN_HOLD_MARGIN_PCT;
361	    if (!holding) {
362	      state.integralAcc += state.error * DT;
363	      const lim = 30 / (state.ki || 0.08);
364	    state.integralAcc = Math.max(-lim, Math.min(lim, state.integralAcc));
365	    }
366	
367	    const derivative = (state.error - state.lastError) / DT;
368	    state.lastError = state.error;
369	
370	    state.pTerm = state.kp * state.error;
371	    state.iTerm = state.ki * state.integralAcc;
372	    state.dTerm = state.kd * derivative;
373	
374	    let computed = state.pTerm + state.iTerm + state.dTerm;
375	    // ensure rawPwmCommand forces 0 when holding:
376	    rawPwmCommand = holding ? 0.0 : Math.max(0.0, Math.min(100.0, computed));
377	  }
378	
379	  // Interlock overrides
380	  state.activeFaults = [];
381	  if (state.interlocks.dryRun) {
382	    rawPwmCommand = 0.0;
383	    state.activeFaults.push('LOCK: RESERVOIR CAVITATION PREVENTED (LEVEL < 5%)');
384	  }
385	  if (state.interlocks.pipeBurst) {
386	    rawPwmCommand = 0.0;
387	    state.activeFaults.push('LOCK: MAJOR HYDRAULIC PIPE RUPTURE DETECTED');
388	  }
389	  if (state.interlocks.floodRisk) {
390	    rawPwmCommand = 0.0;
391	    state.activeFaults.push('LOCK: ANTI-FLOODING / ROOT ASPHYXIATION OVERRIDE');
392	  }
393	  if (state.interlocks.thermalTrip) {
394	    rawPwmCommand = Math.min(rawPwmCommand, 30.0);

13	// 1. خوارزمية حساب متطلبات الري ومنع الهدر المناخي
14	function calculateIrrigationDuty(currentVwc, setpoint, rainProb) {
15	  var error = setpoint - currentVwc;
16	
17	  // في حال وجود احتمالية أمطار تفوق 60% يتم تأجيل الضخ تلقائياً
18	  if (rainProb >= 60) {
19	    return { duty: 0, mode: "AUTONOMOUS_RAIN_HOLD" };
20	  }
21	
22	  // حساب نسبة الضخ حسب فارق الرطوبة
23	  var duty = Math.max(0, Math.min(100, error * 2.2));
24	  return { duty: Math.round(duty), mode: "CLOSED_LOOP_ACTIVE" };
25	}

27	// 2. صمامات الأمان الصناعي وحماية المضخة (ISO 10816 Severity Baseline)
28	function evaluateSafety(telemetry) {
29	  let actions = {
30	    tripPump: false,          // إيقاف المضخة التلقائي في الحالات الخطرة
31	    systemStatus: 'NORMAL',   // NORMAL | WARNING | CRITICAL
32	    alarms: []
33	  };
34	
35	  // أ. حماية من التشغيل الجاف (Dry-Run Protection مع فلترة التأخير الزمني)
36	  // لا نوقف المضخة إلا إذا استمرت بالعمل بحمل حقيقي (>15%) بدون تدفق لمدة 4 ثوانٍ متتالية
37	  if (telemetry.pumpState && telemetry.pumpDuty > 15.0 && telemetry.flowRate <= 0.8) {
38	    dryRunDebounceCounter++;
39	    if (dryRunDebounceCounter >= 4) {
40	      actions.tripPump = true;
41	      actions.systemStatus = 'CRITICAL';
42	      actions.alarms.push('CRITICAL: Dry-run verified. Zero flow while pump running under load.');
43	    } else {
44	      actions.systemStatus = 'WARNING';
45	      actions.alarms.push(`WARNING: Establishing hydraulic flow (${dryRunDebounceCounter}/4s)...`);
46	    }
47	  } else {
48	    dryRunDebounceCounter = Math.max(0, dryRunDebounceCounter - 1);
49	  }
50	
51	  // ب. حماية المحرك من ارتفاع درجة الحرارة (Thermal Overheat)
52	  if (telemetry.motorTemp >= 85.0) {
53	    actions.tripPump = true;
54	    actions.systemStatus = 'CRITICAL';
55	    actions.alarms.push(`CRITICAL: Motor overheat (${telemetry.motorTemp}°C >= 85°C).`);
56	  } else if (telemetry.motorTemp >= 75.0) {
57	    actions.systemStatus = actions.systemStatus === 'CRITICAL' ? 'CRITICAL' : 'WARNING';
58	    actions.alarms.push(`WARNING: High motor temperature (${telemetry.motorTemp}°C).`);
59	  }
60	
61	  // ج. معيار الاهتزاز الميكانيكي وتآكل المحامل (ISO 10816 Class II)
62	  // Class II thresholds: Warning >= 2.8 mm/s, Critical Trip >= 4.5 mm/s (ISO 10816-3 Group 2 Rigid C/D boundary)
63	  if (telemetry.vibrationRms >= 4.5) {
64	    actions.tripPump = true;
65	    actions.systemStatus = 'CRITICAL';
66	    actions.alarms.push(`CRITICAL: Unacceptable vibration (${telemetry.vibrationRms} mm/s). Emergency stop.`);
67	  } else if (telemetry.vibrationRms >= 2.8) {
68	    actions.systemStatus = actions.systemStatus === 'CRITICAL' ? 'CRITICAL' : 'WARNING';
69	    actions.alarms.push(`WARNING: Unsatisfactory vibration (${telemetry.vibrationRms} mm/s). Maintenance needed.`);
70	  }
71	
72	  return actions;
73	}

85	def receive_vibration(data: VibrationPayload):
86	    x = np.asarray(data.vibrationWaveform, dtype=float)
87	    n, fs = len(x), data.samplingRateHz
88	
89	    # Strip DC component
90	    x = x - x.mean()
91	    vibration_rms = float(np.sqrt(np.mean(x**2)))
92	
93	    # Hanning window
94	    w = np.hanning(n)
95	    spectrum = 2.0 * np.abs(rfft(x * w)) / w.sum()
96	    spectrum[0] = 0.0
97	    if n % 2 == 0:
98	        spectrum[-1] /= 2
99	
100	    frequencies = rfftfreq(n, d=1 / fs)
101	
102	    global latest_vibration, latest_fft
103	    latest_vibration = x.tolist()
104	    latest_fft = {
105	        "frequencies_hz": [round(float(f), 2) for f in frequencies],
106	        "magnitudes": [round(float(m), 4) for m in spectrum],
107	    }
108	
109	    # Dominant frequency
110	    dominant_index = int(np.argmax(spectrum))
111	    dominant_frequency = float(frequencies[dominant_index])
112	    dominant_amplitude = float(spectrum[dominant_index])
113	
114	    # 1x rotational component in 30-70 Hz band
115	    run = (frequencies >= 30) & (frequencies <= 70)
116	    one_x_idx = int(np.argmax(np.where(run, spectrum, 0))) if np.any(run) else 0
117	    one_x_frequency = float(frequencies[one_x_idx])
118	    one_x_amplitude = float(spectrum[one_x_idx])
119	
120	    # Bearing-region harmonics tracking at 3.56x shaft speed
121	    bearing_frequency_target = 3.56 * one_x_frequency
122	    band = (frequencies >= 0.95 * bearing_frequency_target) & (frequencies <= 1.05 * bearing_frequency_target)
123	
124	    # Bearing band RMS
125	    bearing_band_low = max(0.95 * bearing_frequency_target - 5, 0)
126	    bearing_band_high = min(1.05 * bearing_frequency_target + 5, fs / 2)
127	    bearing_band_mask = (frequencies >= bearing_band_low) & (frequencies <= bearing_band_high)
128	    if np.any(bearing_band_mask):
129	        bearing_band_rms = float(np.sqrt(np.mean(spectrum[bearing_band_mask] ** 2)))
130	    else:
131	        bearing_band_rms = 0.0
132	
133	    # 2x rotational
134	    target_2x = one_x_frequency * 2
135	    two_x_index = int(np.argmin(np.abs(frequencies - target_2x)))
136	    two_x_frequency = float(frequencies[two_x_index])
137	    two_x_amplitude = float(spectrum[two_x_index])
138	
139	    # Bearing frequency at target
140	    bearing_index = int(np.argmin(np.abs(frequencies - bearing_frequency_target)))
141	    bearing_frequency = float(frequencies[bearing_index])
142	    bearing_amplitude = float(spectrum[bearing_index])
143	
144	    # Amplitude ratios
145	    two_x_to_one_x_ratio = two_x_amplitude / one_x_amplitude if one_x_amplitude > 0 else 0.0
146	    one_x_to_rms_ratio = one_x_amplitude / vibration_rms if vibration_rms > 0 else 0.0
147	    bearing_to_one_x_ratio = bearing_amplitude / one_x_amplitude if one_x_amplitude > 0 else 0.0
148	    bearing_band_to_rms_ratio = bearing_band_rms / vibration_rms if vibration_rms > 0 else 0.0
149	
150	    # Spectral stats
151	    mean_spectrum = float(np.mean(spectrum))
152	    one_x_to_mean_spectrum_ratio = one_x_amplitude / mean_spectrum if mean_spectrum > 0 else 0.0
153	    two_x_to_mean_spectrum_ratio = two_x_amplitude / mean_spectrum if mean_spectrum > 0 else 0.0
154	    spectral_energy = float(np.sum(spectrum**2))
155	    total_spectrum = float(np.sum(spectrum))
156	
157	    if total_spectrum > 0:
158	        spectral_centroid = float(np.sum(frequencies * spectrum) / total_spectrum)
159	        spectral_bandwidth = float(
160	            np.sqrt(np.sum(((frequencies - spectral_centroid) ** 2) * spectrum) / total_spectrum)
161	        )
162	    else:
163	        spectral_centroid = 0.0
164	        spectral_bandwidth = 0.0
165	
166	    # Real Peak Separation using find_peaks (sorted by descending magnitude)
167	    peak_idx, _ = find_peaks(spectrum, distance=3)
168	    top_peak_indices = peak_idx[np.argsort(spectrum[peak_idx])[::-1][:5]] if len(peak_idx) else np.array([], dtype=int)
169	    peaks = [
170	        {
171	            "frequency_hz": round(float(frequencies[p]), 2),
172	            "magnitude": round(float(spectrum[p]), 4),
173	        }
174	        for p in top_peak_indices
175	    ]
176	
177	    if DEBUG:
178	        print("\n========== VIBRATION FEATURES ==========")
179	        print(f"RMS: {vibration_rms:.4f}")
180	        print(f"Dominant frequency: {dominant_frequency:.2f} Hz")

152	    if (audioMuted || !audioCtx) return;
153	
154	    try {
155	      var osc = audioCtx.createOscillator();
156	      var gain = audioCtx.createGain();
157	
158	      osc.type = type || "sine";
159	      osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
160	
161	      gain.gain.setValueAtTime(
162	        vol || 0.1,
163	        audioCtx.currentTime
164	      );
165	
166	      gain.gain.exponentialRampToValueAtTime(
167	        0.001,
168	        audioCtx.currentTime + duration
169	      );
170	
171	      osc.connect(gain);
172	      gain.connect(audioCtx.destination);
173	
174	      osc.start();
175	      osc.stop(audioCtx.currentTime + duration);
176	    } catch (e) {}

2476	    var textDecoder =
2477	      new TextDecoderStream();
2478	
2479	    usbPort.readable.pipeTo(
2480	      textDecoder.writable
2481	    );

2483	    var reader =
2484	      textDecoder.readable.getReader();
2485	
2486	    usbReader =
2487	      reader;
2488	
2489	    try {
2490	      while (true) {
2491	        var result =
2492	          await reader.read();

2494	        if (result.done) {
2495	          break;
2496	        }
2497	
2498	        if (result.value) {
2499	          serialLineBuffer +=
2500	            result.value;

2504	              "\n"
2505	            );

2508	            lines.pop();

2510	          for (
2511	            var i = 0;
2512	            i < lines.length;
2513	            i++
2514	          ) {
2515	            var line =
2516	              lines[i].trim();
2517	
2518	            if (
2519	              line.startsWith("{") &&
2520	              line.endsWith("}")

2522	              parseHardwarePacket(

2532	        "[USB DISCONNECT]</strong> " +

78	  // حساب الـ Spectrum مع معايرة السعة الفيزيائية الصحيحة (Single-Sided Scaled DFT)
79	  function computeFFT(samples) {
80	    var N = samples.length;
81	    var magnitudes = [];
82	    var halfN = Math.floor(N / 2);
83	
84	    for (var k = 0; k < halfN; k++) {
85	      var real = 0;
86	      var imag = 0;
87	      for (var n = 0; n < N; n++) {
88	        var angle = (2 * Math.PI * k * n) / N;
89	        real += samples[n] * Math.cos(angle);
90	        imag -= samples[n] * Math.sin(angle);
91	      }
92	      var mag = Math.sqrt(real * real + imag * imag);
93	      // التردد الصفري DC يقسم على N، وبقية الترددات تضرب في 2/N لمعايرة السعة الفيزيائية الحقيقية
94	      var scaled = (k === 0) ? (mag / N) : ((2 * mag) / N);
95	      magnitudes.push(Number(scaled.toFixed(4)));
96	    }
97	    return magnitudes;
98	  }
